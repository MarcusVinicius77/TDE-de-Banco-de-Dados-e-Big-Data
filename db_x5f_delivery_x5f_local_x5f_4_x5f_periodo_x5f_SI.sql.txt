DROP DATABASE IF EXISTS db_delivery_local;
CREATE DATABASE db_delivery_local;
USE db_delivery_local;

-- =================================================================
-- ITEM 1: CRIAÇÃO DAS TABELAS RELACIONADAS
-- =================================================================

-- TABELA 1: Clientes (Quem pede) - PK id_cliente
CREATE TABLE Clientes (
    id_cliente INT AUTO_INCREMENT PRIMARY KEY, -- PK
    nome VARCHAR(100) NOT NULL,
    telefone VARCHAR(20),
    bairro VARCHAR(50),
    endereco TEXT
);

-- TABELA 2: Restaurantes (Quem vende) - PK id_restaurante
CREATE TABLE Restaurantes (
    id_restaurante INT AUTO_INCREMENT PRIMARY KEY, -- PK
    nome_restaurante VARCHAR(100) NOT NULL,
    tipo_comida VARCHAR(50),
    taxa_entrega DECIMAL(5,2)
);

-- TABELA 3: Produtos (Cardápio, depende do restaurante)
-- FK: id_restaurante liga produto ao restaurante
CREATE TABLE Produtos (
    id_produto INT AUTO_INCREMENT PRIMARY KEY, -- PK
    nome_produto VARCHAR(100) NOT NULL,
    preco DECIMAL(6,2) NOT NULL,
    id_restaurante INT NOT NULL, -- FK - De qual restaurante é esse produto
    FOREIGN KEY (id_restaurante) REFERENCES Restaurantes(id_restaurante) ON DELETE CASCADE
);

-- TABELA 4: Pedidos (Tabela principal da INTEGRAÇÃO)
-- Liga Cliente com Restaurante - EXATAMENTE o que o líder pediu
CREATE TABLE Pedidos (
    id_pedido INT AUTO_INCREMENT PRIMARY KEY, -- PK do pedido
    id_cliente INT NOT NULL, -- FK 1 - Quem pediu
    id_restaurante INT NOT NULL, -- FK 2 - De onde pediu
    data_pedido DATETIME DEFAULT CURRENT_TIMESTAMP,
    status VARCHAR(20) DEFAULT 'Preparando',
    FOREIGN KEY (id_cliente) REFERENCES Clientes(id_cliente) ON DELETE CASCADE,
    FOREIGN KEY (id_restaurante) REFERENCES Restaurantes(id_restaurante) ON DELETE CASCADE
);

-- TABELA 5: Itens_Pedido (Detalhe do pedido - N:N entre Pedidos e Produtos)
CREATE TABLE Itens_Pedido (
    id_item INT AUTO_INCREMENT PRIMARY KEY,
    id_pedido INT NOT NULL, -- FK para Pedido
    id_produto INT NOT NULL, -- FK para Produto
    quantidade INT NOT NULL,
    FOREIGN KEY (id_pedido) REFERENCES Pedidos(id_pedido) ON DELETE CASCADE,
    FOREIGN KEY (id_produto) REFERENCES Produtos(id_produto) ON DELETE CASCADE
);

-- =================================================================
-- INTEGRAÇÃO COM JOIN - O QUE O LÍDER PEDIU
-- =================================================================

-- SELECT 1: Mostra Pedido com nome do Cliente + nome do Restaurante
SELECT 
    p.id_pedido,
    c.nome AS Cliente,
    r.nome_restaurante AS Restaurante,
    p.status
FROM Pedidos p
INNER JOIN Clientes c ON p.id_cliente = c.id_cliente
INNER JOIN Restaurantes r ON p.id_restaurante = r.id_restaurante;

-- SELECT 2: INTEGRAÇÃO COMPLETA - Mostra tudo
SELECT 
    cli.nome AS Cliente,
    rest.nome_restaurante AS Restaurante,
    prod.nome_produto AS Produto,
    ip.quantidade,
    prod.preco,
    (ip.quantidade * prod.preco) AS Total_Item
FROM Pedidos ped
JOIN Clientes cli ON ped.id_cliente = cli.id_cliente
JOIN Restaurantes rest ON ped.id_restaurante = rest.id_restaurante
JOIN Itens_Pedido ip ON ped.id_pedido = ip.id_pedido
JOIN Produtos prod ON ip.id_produto = prod.id_produto
ORDER BY ped.id_pedido;
